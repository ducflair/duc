1//! Schema bootstrap — applies `schema/duc.sql` to a fresh connection.
2//!
3//! Called by both `native` and `wasm` backends after opening a connection.
4//! The canonical schema lives at `duc.sql` in the workspace root; it is
5//! embedded at compile time so the binary carries no file-system dependency.
6
7use rusqlite::Connection;
8
9use crate::db::DbError;
10
11/// Canonical schemas, embedded at compile time.
12/// `build.rs` copies them into `OUT_DIR` so this works both in-tree and from sdists.
13/// Applied in order: duc.sql → version_control.sql → search.sql
14const DUC_SCHEMA: &str = include_str!(concat!(env!("OUT_DIR"), "/duc.sql"));
15const VERSION_CONTROL_SCHEMA: &str = include_str!(concat!(env!("OUT_DIR"), "/version_control.sql"));
16const SEARCH_SCHEMA: &str = include_str!(concat!(env!("OUT_DIR"), "/search.sql"));
17
18/// Expected `application_id` written by `duc.sql`.
19const APP_ID: i64 = 1_146_569_567; // "DUC_" in ASCII
20
21/// Schema version read from `duc.sql` at compile time by `build.rs`.
22/// Never hardcode this — it is always in sync with the SQL file.
23const CURRENT_VERSION: i64 = {
24    // build.rs emits DUC_SCHEMA_USER_VERSION as a decimal integer string.
25    let bytes = env!("DUC_SCHEMA_USER_VERSION").as_bytes();
26    let mut val: i64 = 0;
27    let mut i = 0;
28    while i < bytes.len() {
29        val = val * 10 + (bytes[i] - b'0') as i64;
30        i += 1;
31    }
32    val
33};
34
35/// Runtime accessor for the compile-time schema version integer.
36pub fn current_schema_version_int() -> i64 {
37    CURRENT_VERSION
38}
39
40// Migration registry auto-generated by build.rs from `schema/migrations/*.sql`.
41// Each entry is (from_version, to_version, sql). Sorted ascending by from_version.
42include!(concat!(env!("OUT_DIR"), "/migrations_registry.rs"));

46/// Current schema version as a semver string (generated at build time).
47pub const CURRENT_SCHEMA_VERSION_SEMVER: &str = env!("DUC_SCHEMA_VERSION");

49pub const DUC_SCHEMA_SQL: &str = DUC_SCHEMA;
50pub const VERSION_CONTROL_SCHEMA_SQL: &str = VERSION_CONTROL_SCHEMA;
51pub const SEARCH_SCHEMA_SQL: &str = SEARCH_SCHEMA;

53/// Per-connection PRAGMAs that must be re-applied on every open (they are
54/// either ephemeral or take effect only after the first statement on the
55/// connection).
56#[cfg(not(all(target_family = "wasm", target_os = "unknown")))]
57const CONN_PRAGMAS: &str = "
58    PRAGMA journal_mode = WAL;
59    PRAGMA foreign_keys  = ON;
60";
61
62#[cfg(all(target_family = "wasm", target_os = "unknown"))]
63const CONN_PRAGMAS: &str = "
64    PRAGMA journal_mode = MEMORY;
65    PRAGMA foreign_keys  = ON;
66";
67
68/// Apply the full schema to `conn` if it is a new (empty) database, then
69/// always re-apply the per-connection PRAGMAs.
70///
71/// Safe to call multiple times on an already-bootstrapped database — it
72/// will only run the DDL once.
73pub(crate) fn bootstrap(conn: &Connection) -> Result<(), DbError> {
74    let user_version: i64 = conn.pragma_query_value(None, "user_version", |r| r.get(0))?;
75
76    if user_version == 0 {
77        // Apply schemas in order.
78        conn.execute_batch(DUC_SCHEMA)
79            .map_err(|e| DbError::Bootstrap(format!("duc.sql apply failed: {e}")))?;
80        conn.execute_batch(VERSION_CONTROL_SCHEMA)
81            .map_err(|e| DbError::Bootstrap(format!("version_control.sql apply failed: {e}")))?;
82        conn.execute_batch(SEARCH_SCHEMA)
83            .map_err(|e| DbError::Bootstrap(format!("search.sql apply failed: {e}")))?;
84
85        // Verify the schema set the expected application_id.
86        let app_id: i64 = conn.pragma_query_value(None, "application_id", |r| r.get(0))?;
87        if app_id != APP_ID {
88            return Err(DbError::Bootstrap(format!(
89                "unexpected application_id after bootstrap: {app_id} (expected {APP_ID})"
90            )));
91        }
92    } else if user_version == CURRENT_VERSION {
93        // Current schema — just ensure per-connection pragmas are active.
94        conn.execute_batch(CONN_PRAGMAS)
95            .map_err(|e| DbError::Bootstrap(format!("pragma apply failed: {e}")))?;
96    } else {
97        // Older database: normalize any known schema drift before running the
98        // canonical migration chain. Fixtures written by prerelease code may
99        // have applied parts of later schemas without bumping user_version, so
100        // we add columns/indexes migrations expect when missing and recreate
101        // objects that already exist to keep the migration SQL idempotent.
102        normalize_legacy_schema(conn, user_version)?;
103
104        // Walk the migration chain until we reach CURRENT_VERSION.
105        // build.rs generates MIGRATIONS sorted by from_version, so chaining
106        // (e.g. 3000000→3000001→3000002) works without any code changes here.
107        let mut current = user_version;
108        loop {
109            match MIGRATIONS.iter().find(|(from, _, _)| *from == current) {
110                Some((from, to, sql)) => {
111                    if let Err(error) = conn.execute_batch(sql) {
112                        let _ = conn.execute_batch("ROLLBACK");
113                        return Err(DbError::Bootstrap(format!(
114                            "migration {from}\u{2192}{to} failed: {error}"
115                        )));
116                    }
117                    // Re-read the version the migration SQL set via PRAGMA user_version.
118                    current = conn.pragma_query_value(None, "user_version", |r| r.get(0))?;
119                    if current == CURRENT_VERSION {
120                        break;
121                    }
122                }
123                None => {
124                    return Err(DbError::Bootstrap(format!(
125                        "unsupported schema version {user_version}; expected 0 or {CURRENT_VERSION}"
126                    )));
127                }
128            }
129        }
130        conn.execute_batch(CONN_PRAGMAS)
131            .map_err(|e| DbError::Bootstrap(format!("pragma apply failed: {e}")))?;
132    }
133
134    Ok(())
135}

137/// Add missing columns/objects that later migrations expect.
138///
139/// Some test fixtures were produced by intermediate code that created tables
140/// such as `search_elements` or renamed `element_model.svg_path` to `thumbnail`
141/// while still reporting an older `user_version`. Adding the missing legacy
142/// columns back lets the canonical migration SQL run without modification.
143fn normalize_legacy_schema(conn: &Connection, user_version: i64) -> Result<(), DbError> {
144    // 3000003→3000004 migration expects `element_model.svg_path`.
145    if user_version <= 3000004 {
146        let has_column: bool = conn
147            .query_row(
148                "SELECT 1 FROM pragma_table_info('element_model') WHERE name = 'svg_path'",
149                [],
150                |_| Ok(true),
151            )
152            .unwrap_or(false);
153        if !has_column {
154            conn.execute(
155                "ALTER TABLE element_model ADD COLUMN svg_path TEXT DEFAULT NULL",
156                [],
157            )
158            .map_err(|e| {
159                DbError::Bootstrap(format!(
160                    "legacy normalization: add element_model.svg_path failed: {e}"
161                ))
162            })?;
163        }
164    }
165
166    Ok(())
167}
